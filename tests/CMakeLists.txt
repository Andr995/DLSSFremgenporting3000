cmake_minimum_required(VERSION 3.24)
project(MfgRegressionTests LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
set(STREAMLINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../deps/Streamline" CACHE PATH "Streamline SDK")
set(NATIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../source/native")

add_executable(Fp8ReferenceTests fp8_reference_tests.cpp "${NATIVE_ROOT}/ampere_fp8_emulator.cpp")
add_executable(PolicyTests policy_tests.cpp)
add_executable(ControlConfigTests control_config_tests.cpp "${NATIVE_ROOT}/control_config.cpp")
add_executable(StatusValidationTests status_validation_tests.cpp "${NATIVE_ROOT}/status_validation.cpp")
target_include_directories(ControlConfigTests SYSTEM PRIVATE "${STREAMLINE_ROOT}/external/json/include")
target_include_directories(StatusValidationTests SYSTEM PRIVATE "${STREAMLINE_ROOT}/external/json/include")
set(REGRESSION_TARGETS Fp8ReferenceTests PolicyTests ControlConfigTests StatusValidationTests)
if(WIN32)
    add_executable(RuntimePathsTests runtime_paths_tests.cpp "${NATIVE_ROOT}/runtime_paths.cpp")
    add_library(LifetimeFixture SHARED lifetime_fixture.cpp)
    target_compile_features(LifetimeFixture PRIVATE cxx_std_20)
    target_include_directories(LifetimeFixture PRIVATE "${NATIVE_ROOT}")
    add_executable(ModuleLifetimeTests module_lifetime_tests.cpp)
    add_dependencies(ModuleLifetimeTests LifetimeFixture)
    list(APPEND REGRESSION_TARGETS RuntimePathsTests ModuleLifetimeTests)
endif()
foreach(target ${REGRESSION_TARGETS})
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${NATIVE_ROOT}")
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(WIN32)
        target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(MINGW)
            target_link_options(${target} PRIVATE -static)
        endif()
    endif()
    if(target STREQUAL "ModuleLifetimeTests")
        add_test(NAME ${target} COMMAND ${target} "$<TARGET_FILE:LifetimeFixture>")
    else()
        add_test(NAME ${target} COMMAND ${target})
    endif()
    set_tests_properties(${target} PROPERTIES TIMEOUT 60)
endforeach()
